
    const closeBtn = card.querySelector("#close-card")
    if (closeBtn) {
      closeBtn.addEventListener("click", () => this.hideExhibitCard())
    }
  }

  showExhibitCard(exhibit: any) {
    if (!this.exhibitCard) return
    const title = this.exhibitCard.querySelector("#card-title")
    const author = this.exhibitCard.querySelector("#card-author")
    const size = this.exhibitCard.querySelector("#card-size")
    const desc = this.exhibitCard.querySelector("#card-desc")
    if (title) title.textContent = exhibit.title
    if (author) author.textContent = "作者: " + exhibit.author
    if (size) size.textContent = "尺寸: " + exhibit.size
    if (desc) desc.textContent = exhibit.description
    this.exhibitCard.style.opacity = "1"
    this.exhibitCard.style.transform = "translate(-50%,-50%) scale(1)"
    this.exhibitCard.style.pointerEvents = "auto"
  }
